class Solution {
    public int removeElement(int[] nums, int val) {
        int k = 0; // Pointer for the position of non-val elements
        for (int i = 0; i < nums.length; i++) {
            if (nums[i] != val) {
                nums[k] = nums[i]; // Place non-val element at position k
                k++; // Increment the pointer
            }
        }
        return k; // k represents the length of the modified array
    }
}
