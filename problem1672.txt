class Solution {
    public int maximumWealth(int[][] accounts) {
        int maxwealth = 0;
        
        for(int[] account : accounts)
        {
            int sum= 0;
            for(int wealth : account)
            {
                sum= sum + wealth;
            }
            maxwealth = Math.max(maxwealth, sum);
        }
        return maxwealth;
    }
}
