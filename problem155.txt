class MinStack {

    int top;
    List<Integer> arrlist = new ArrayList<>();
    public MinStack() {
    
        this.top = -1;
    }
    
    public void push(int val) {
        top++;
        arrlist.add(top,val);
        
        
    }
    
    public void pop() {
        arrlist.remove(top);
        top--;
    }
    
    public int top() {
        return arrlist.get(top);
    }
    
    public int getMin() {
        int min = Integer. MAX_VALUE;
        for(int i : arrlist){
            if (i < min){
                min = i;
            }
        }
        return min;
        
    }
}
